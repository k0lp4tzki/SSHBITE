# Scripts used:
## sudomail4.py
import asyncio, sys
sys.path.insert(0, "/workspace")
from tests.e2e.lab import LabControl
lc = LabControl({"b5": ["172.30.2.15"], "a4": ["172.30.1.14"]})
on = r"""
printf '#!/bin/sh\n{ echo "args: $*"; cat; echo ----; } >> /tmp/p3mail.txt\n' > /run/fakemail; chmod 755 /run/fakemail
printf 'Defaults mailerpath=/run/fakemail\n' > /etc/sudoers.d/zzp3mail; chmod 440 /etc/sudoers.d/zzp3mail
"""
off = r"""
sleep 2; echo "== mail captured:"; cat /tmp/p3mail.txt 2>/dev/null || echo "(none)"
rm -f /etc/sudoers.d/zzp3mail /run/fakemail /tmp/p3mail.txt; ls /etc/sudoers.d
"""
def as_sshbite(label, cmd):
    return f'echo "== {label}:"; su sshbite -s /bin/sh -c "{cmd}; echo rc=\\$?" 2>&1\n'
a4 = on + as_sshbite("a4 (rule needs password) sudo -n", "sudo -n /bin/true") \
        + as_sshbite("a4 wrong password", "printf 'wrong\\n' | sudo -S -k -p \\'\\' /bin/true") + off
b5 = on + "cp -p /etc/shadow /run/shadow.orig; echo 'sshbite:p3test-pw' | chpasswd\n" \
        + as_sshbite("b5 (no rule) sudo -n", "sudo -n /bin/true") \
        + "sleep 1; echo '-- mail so far:'; cat /tmp/p3mail.txt 2>/dev/null || echo '(none)'\n" \
        + as_sshbite("b5 (no rule) correct password", "printf 'p3test-pw\\n' | sudo -S -k -p \\'\\' /bin/true") \
        + "cat /run/shadow.orig > /etc/shadow && rm /run/shadow.orig\n" + off
print("######## a4"); print(asyncio.run(lc._run("a4", a4)))
print("######## b5"); print(asyncio.run(lc._run("b5", b5)))

import asyncio, sys

printf 'Defaults mailerpath=/run/fakemail\n' > /etc/sudoers.d/zzp3mail; chmod 440 /etc/sudoers.d/zzp3mail
cat /etc/sudoers.d/sshbite

sleep 2; echo "== mail captured:"; cat /tmp/p3mail.txt 2>/dev/null || echo "(none)"

"""
